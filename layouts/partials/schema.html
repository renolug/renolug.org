{{/* schema.org JSON-LD. Emits an Organization node on the home page and an Event node on any page carrying an `event` map in its front matter (see content/meetings/). Built as Go maps and passed through jsonify so the output is always valid JSON. */}} {{- $nodes := slice -}} {{- if .IsHome -}} {{- $sameAs := slice -}} {{- range $key, $url := site.Params.links -}} {{- if and $url (hasPrefix $url "http") -}} {{- $sameAs = $sameAs | append $url -}} {{- end -}} {{- end -}} {{- $org := dict "@type" "Organization" "@id" (printf "%s#organization" site.BaseURL) "name" site.Title "alternateName" "RLUG" "url" site.BaseURL "description" site.Params.description "logo" ("/img/og.png" | absURL) -}} {{- with $sameAs }}{{ $org = merge $org (dict "sameAs" .) }}{{ end -}} {{- $nodes = $nodes | append $org -}} {{- end -}} {{- $page := . -}} {{- with .Params.event -}} {{- $address := dict "@type" "PostalAddress" "addressLocality" .locality "addressRegion" .region "addressCountry" "US" -}} {{- with .street }}{{ $address = merge $address (dict "streetAddress" .) }}{{ end -}} {{- $event := dict "@type" "Event" "name" (printf "%s: %s" site.Title $page.Title) "url" $page.Permalink "description" ($page.Description | plainify | chomp) "startDate" ($page.Date.Format "2006-01-02T15:04:05-07:00") "endDate" (or .end ($page.Date.Format "2006-01-02T15:04:05-07:00")) "eventStatus" "https://schema.org/EventScheduled" "eventAttendanceMode" "https://schema.org/OfflineEventAttendanceMode" "image" ("/img/og.png" | absURL) "location" (dict "@type" "Place" "name" .venue "address" $address) "organizer" (dict "@type" "Organization" "name" site.Title "url" site.BaseURL) "offers" (dict "@type" "Offer" "price" "0" "priceCurrency" "USD" "availability" "https://schema.org/InStock" "url" (or .register $page.Permalink) ) -}} {{- with .speaker -}} {{- $performer := dict "@type" "Person" "name" . -}} {{- with $page.Params.event.speakerOrg }}{{ $performer = merge $performer (dict "affiliation" (dict "@type" "Organization" "name" .)) }}{{ end -}} {{- $event = merge $event (dict "performer" $performer) -}} {{- end -}} {{- $nodes = $nodes | append $event -}} {{- end -}} {{- with $nodes }} {{/* safeJS: jsonify already escapes <, > and & as \uXXXX, so this can't break out of the script tag. */}} {{- end -}}
